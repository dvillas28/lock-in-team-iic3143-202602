\begin{frame}{Anexo: CU-01 Elegir institución}
\small
\textbf{Actor:} Usuario autenticado\par
\begin{columns}[T,onlytextwidth]\begin{column}{0.60\textwidth}
\textbf{Flujo básico}\begin{enumerate}
\item Solicitar las instituciones accesibles.
\item Mostrar solo memberships activas.
\item Elegir una institución visible.
\item Establecer contexto de navegación, sin asignar roles.\end{enumerate}\end{column}\begin{column}{0.37\textwidth}
\textbf{Meta}\par seleccionar un contexto institucional válido para continuar sus actividades.
\par\vspace{0.2cm}\textbf{Alternativas}\par \begin{itemize}
\item Sin instituciones accesibles
\item La pertenencia deja de estar activa\end{itemize}\end{column}\end{columns}\par\vspace{0.15cm}
{\footnotesize Precondición: identidad autenticada vigente. Ficha íntegra: 03-use-cases-and-requirements.md.}
\end{frame}

\begin{frame}{Anexo: CU-02 Entrar al contexto}
\small
\textbf{Actor:} Usuario autenticado\par
\begin{columns}[T,onlytextwidth]\begin{column}{0.60\textwidth}
\textbf{Flujo básico}\begin{enumerate}
\item Solicitar entrar al contexto elegido.
\item Resolver Institution y comprobar visibilidad.
\item Validar membership y recuperar roles de curso/sección.
\item Presentar identidad contextual y vistas permitidas.\end{enumerate}\end{column}\begin{column}{0.37\textwidth}
\textbf{Meta}\par trabajar únicamente dentro de un contexto al que pertenece, sin confundir pertenencia institucional con autorización de curso.
\par\vspace{0.2cm}\textbf{Alternativas}\par \begin{itemize}
\item Institución inexistente o no visible
\item Pertenencia inactiva\end{itemize}\end{column}\end{columns}\par\vspace{0.15cm}
{\footnotesize Precondición transversal: JWT y InstitutionMembership activa. Ficha íntegra: 03-use-cases-and-requirements.md.}
\end{frame}

\begin{frame}{Anexo: CU-03 Ver cursos y secciones}
\small
\textbf{Actor:} Coordinador de curso / Docente de sección / Ayudante de sección / Estudiante\par
\begin{columns}[T,onlytextwidth]\begin{column}{0.60\textwidth}
\textbf{Flujo básico}\begin{enumerate}
\item Solicitar cursos propios en la institución.
\item Listar cursos con pertenencia académica activa.
\item Elegir curso y pedir sus secciones.
\item Mostrar todo al coordinador y solo secciones propias al resto.
\item Abrir una sección visible.\end{enumerate}\end{column}\begin{column}{0.37\textwidth}
\textbf{Meta}\par encontrar los cursos y secciones donde mantiene un rol activo, con visibilidad ajustada a su alcance.
\par\vspace{0.2cm}\textbf{Alternativas}\par \begin{itemize}
\item Sin cursos visibles
\item Curso o sección fuera del alcance\end{itemize}\end{column}\end{columns}\par\vspace{0.15cm}
{\footnotesize Precondición transversal: JWT y InstitutionMembership activa. Ficha íntegra: 03-use-cases-and-requirements.md.}
\end{frame}

\begin{frame}{Anexo: CU-04 Secciones y roles}
\small
\textbf{Actor:} Coordinador de curso\par
\begin{columns}[T,onlytextwidth]\begin{column}{0.60\textwidth}
\textbf{Flujo básico}\begin{enumerate}
\item Consultar secciones y participantes del curso.
\item Definir o actualizar una sección.
\item Buscar una persona de la misma Institution.
\item Validar curso, sección y pertenencia institucional.
\item Asignar teacher, assistant o student.
\item Registrar cambios sensibles con auditoría.\end{enumerate}\end{column}\begin{column}{0.37\textwidth}
\textbf{Meta}\par disponer de secciones actualizadas y asignar roles académicos a participantes de la misma institución.
\par\vspace{0.2cm}\textbf{Alternativas}\par \begin{itemize}
\item Falta de rol coordinador
\item Participante o relación fuera de la institución
\item Rol activo duplicado o datos de sección inválidos
\item Cambiar o desactivar una pertenencia
\item Actualizar configuración general del curso\end{itemize}\end{column}\end{columns}\par\vspace{0.15cm}
{\footnotesize Precondición transversal: JWT y InstitutionMembership activa. Ficha íntegra: 03-use-cases-and-requirements.md.}
\end{frame}

\begin{frame}{Anexo: CU-06 Consultar material}
\small
\textbf{Actor:} Estudiante\par

\begin{frame}{Anexo: cálculo de notas y promedio}
\small
\begin{block}{Puntaje y escala chilena}
\[p=100\,\frac{\mathrm{score\_points}}{\sum\mathrm{question.points}}\]
\[\mathrm{grade\_value}=\begin{cases}1+3p/60 & p\leq60\\4+3(p-60)/40 & p>60\end{cases}\]
\end{block}
\begin{block}{Promedio parcial}
\[\frac{\sum_{\text{notas publicadas}}\mathrm{grade\_value}\cdot\mathrm{weight\_percent}}{\sum_{\text{notas publicadas}}\mathrm{weight\_percent}}\]
\end{block}
\footnotesize HALF\_UP: puntaje porcentual a dos decimales y nota a uno. Sin ponderación publicada: null. El promedio se calcula en backend, sin columna derivada.
\end{frame}
